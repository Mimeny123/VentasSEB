/* creates */

CREATE TABLE usuarios (
    usuario_id INT UNSIGNED not null primary key AUTO_INCREMENT,
    usuario VARCHAR(50),
    nombre VARCHAR(50),
    apellido VARCHAR(50),
    edad TINYINT UNSIGNED,
    gmail VARCHAR(100),
    telefono VARCHAR(25),
    contraseña VARCHAR(255),
);

CREATE TABLE motivosReportes (
    motivo_id tinyint not null unsigned PRIMARY KEY,
    nombre text
);

CREATE table posteos ( 
    post_id int unsigned not null AUTO_INCREMENT PRIMARY KEY, 
    titulo VARCHAR(320), 
    descripcion text, 
    precio int unsigned, 
    imagen varchar(400), 
    estado ENUM('activo', 'pausado', 'sin_stock'), 
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP, 
    usuario_id int unsigned, FOREIGN key (usuario_id) REFERENCES usuarios(usuario_id) 
);

CREATE TABLE reportes (
    reporte_id INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    motivo_id tinyint unsigned,
    descripcion TEXT,
    estado VARCHAR(50),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    usuario_id INT UNSIGNED,
    post_id INT UNSIGNED,
    
    FOREIGN KEY (usuario_id) REFERENCES usuarios(usuario_id),
    FOREIGN KEY (post_id) REFERENCES posteos(post_id),
    FOREIGN KEY (motivo_id) REFERENCES motivosReportes(motivo_id)
);
